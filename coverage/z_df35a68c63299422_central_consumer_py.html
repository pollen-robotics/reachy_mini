1"""Python aiortc consumer of the Reachy Mini HF central signaling relay. 

2 

3Counterpart to :mod:`reachy_mini.media.central_signaling_relay` — the 

4daemon-side *producer* — this module is the *consumer*: a hardware-free 

5client that subscribes to the central HF Space at 

6``https://pollen-robotics-reachy-mini-central.hf.space``, negotiates 

7WebRTC with one robot, and exposes: 

8 

9* decoded **video** frames as numpy arrays (:meth:`latest_frame`); 

10* bidirectional **audio** — the robot mic decoded to mono float32 via an 

11 ``on_pcm`` callback, and an outbound ``out_track`` whose audio plays out 

12 the robot speaker (the daemon advertises audio as ``sendrecv``); 

13* a thread-safe ``send_command`` for the ``data`` channel the daemon offers, 

14 plus an ``on_command_ready`` hook for one-shot setup once it opens. 

15 

16All audio/command extras are optional — with none of them set this behaves 

17exactly as the original video-only consumer. 

18 

19Use this from any cloud backend (HF Space, Cloud Run, etc.) that needs to 

20read a Reachy Mini's camera and/or drive the robot via the visitor's HF 

21token, without instantiating :class:`reachy_mini.ReachyMini` (which 

22requires daemon access and the full hardware stack). 

23 

24Per-process / per-session: instantiate one per visitor (so the visitor's 

25short-lived HF Bearer token scopes the listener to robots they actually 

26own). The instance maintains a long-lived SSE channel to central plus a 

27single :class:`~aiortc.RTCPeerConnection` to one robot. 

28 

29Example:: 

30 

31 from reachy_mini.io.protocol import GotoTargetCmd 

32 from reachy_mini.media.central_consumer import ReachyCentralConsumer 

33 

34 consumer = ReachyCentralConsumer( 

35 hf_token=visitor_token, 

36 robot_peer_id=visitor_robot_peer_id, 

37 consumer_label="my-cloud-backend/visitor-1", 

38 ) 

39 await consumer.start() 

40 

41 # Pull the freshest frame whenever you need it (e.g. in your 

42 # inference loop). ``frame_id`` is a monotonically increasing 

43 # counter so you can dedupe without comparing arrays. 

44 snap = consumer.latest_frame() 

45 if snap is not None: 

46 frame_id, rgb = snap 

47 ... 

48 

49 # Drive the robot over its "data" channel. Build the envelope from 

50 # the shared protocol models so a typo is a type error here rather 

51 # than an "Invalid command" reply from the far end. Thread-safe — 

52 # schedules the actual ``RTCDataChannel.send`` on aiortc's loop. 

53 consumer.send_command(GotoTargetCmd(head=[...], duration=0.4)) 

54 

55 await consumer.stop() 

56 

57Protocol summary (matches ``reachy-mini-sdk.js``): 

58 

59* ``GET {central}/events`` — Server-Sent Events. We receive ``welcome``, 

60 ``list``, ``peer`` (SDP offer + remote ICE), ``endSession``. 

61* ``POST {central}/send`` — JSON envelopes. We send ``setPeerStatus``, 

62 ``startSession``, ``peer`` (SDP answer), ``endSession``. 

63 

64The robot creates the offer; we answer. aiortc gathers all local ICE 

65candidates before ``setLocalDescription`` returns (no out-trickle), so 

66the answer SDP is sent once; trickled remote candidates arriving after 

67the offer are buffered and replayed once the remote description is set. 

68 

69Other niceties baked in: 

70 

71* Heartbeat ``setPeerStatus`` every ~10 s (refreshed from the welcome 

72 lease) so central doesn't evict us as a stale listener. 

73* Force-SSE-reconnect when central returns ``"Connect to /events first"`` 

74 on a POST — that's the signature of a stale relay-side peer record. 

75* ``send_command`` marshalled via ``loop.call_soon_threadsafe`` so 

76 callers running in worker threads (e.g. GPU pipelines) can safely 

77 drive the robot. 

78"""

125def _patch_aiortc_dtls_ciphers() -> None: 

126 """Append a cipher the robot's GStreamer ``webrtcsink`` accepts. 

127 

128 aiortc's default DTLS cipher list shares no cipher with the robot 

129 daemon's GStreamer ``webrtcsink``, so DTLS never completes and the 

130 peer connection never reaches "connected". We wrap 

131 ``RTCCertificate._create_ssl_context`` to add 

132 ``ECDHE-RSA-AES128-GCM-SHA256``. 

133 

134 Upstream fix tracked in aiortc PR #1392 

135 (https://github.com/aiortc/aiortc/pull/1392); remove this shim once a 

136 released aiortc negotiates a compatible cipher by default. 

137 """ 

138 try: 

139 from aiortc.rtcdtlstransport import RTCCertificate 

140 except Exception as e: # pragma: no cover 

141 logger.warning("dtls-patch: aiortc import failed: %r", e) 

142 return 

143 if getattr(RTCCertificate, "_reachy_cipher_patched", False): 

144 return 

145 try: 

146 _orig = RTCCertificate._create_ssl_context 

147 except AttributeError as e: 

148 logger.warning("dtls-patch: _create_ssl_context not found: %r", e) 

149 return 

150 ciphers = ( 

151 b"ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-ECDSA-CHACHA20-POLY1305:" 

152 b"ECDHE-ECDSA-AES128-SHA:ECDHE-ECDSA-AES256-SHA:" 

153 b"ECDHE-RSA-AES128-GCM-SHA256" 

154 ) 

155 

156 def _patched(self: Any, srtp_profiles: Any) -> Any: 

157 ctx = _orig(self, srtp_profiles) 

158 try: 

159 ctx.set_cipher_list(ciphers) 

160 except Exception as e: # pragma: no cover 

161 logger.warning("dtls-patch: set_cipher_list failed: %r", e) 

162 return ctx 

163 

164 # Replacing a method and tagging the class are the point of the shim, 

165 # so both ignores are permanent until aiortc ships the upstream fix. 

166 RTCCertificate._create_ssl_context = _patched # type: ignore[method-assign] 

167 RTCCertificate._reachy_cipher_patched = True # type: ignore[attr-defined] 

168 logger.info("DTLS cipher list extended (+ECDHE-RSA-AES128-GCM-SHA256)")

313 def send_command(self, envelope: AnyCommand | dict[str, Any]) -> bool: 

314 """Write ``envelope`` to the robot's data channel as JSON. 

315 

316 Thread-safe: schedules the actual ``send()`` on the event loop that 

317 owns the aiortc peer connection. 

318 

319 Args: 

320 envelope: preferably a command model from 

321 :mod:`reachy_mini.io.protocol` — the daemon validates what 

322 arrives against that same ``AnyCommand`` union, so building 

323 one here gets the wire format checked statically instead of 

324 at the far end of a WebRTC link. A raw dict is still 

325 accepted for frames the union doesn't cover, notably the 

326 JSON-RPC app-control calls that share this channel. 

327 

328 Returns: 

329 True if the send was queued, False if the channel isn't open. 

330 

331 """ 

332 ch = self._cmd_channel 

333 loop = self._cmd_loop 

334 if ch is None or not self._cmd_channel_open or loop is None: 

335 return False 

336 # `exclude_none` keeps optional fields off the wire; every nullable 

337 # field in the union defaults to None, so the daemon parses either 

338 # form identically. Worth it on set_full_target at control rate. 

339 payload = ( 

340 envelope.model_dump_json(exclude_none=True) 

341 if isinstance(envelope, BaseModel) 

342 else json.dumps(envelope) 

343 ) 

344 

345 def _do_send() -> None: 

346 try: 

347 ch.send(payload) 

348 except Exception as e: 

349 self._cmd_send_errors += 1 

350 if self._cmd_send_errors <= 3: 

351 logger.warning( 

352 "send_command failed (consumer=%s): %r", 

353 self._consumer_label, 

354 e, 

355 ) 

356 

357 try: 

358 loop.call_soon_threadsafe(_do_send) 

359 self._cmd_sent += 1 

360 return True 

361 except RuntimeError: 

362 # Loop closed mid-send; treat as channel closed. 

363 return False

422 async def _run_forever(self) -> None: 

423 backoff = 1.0 

424 while not self._stopping: 

425 try: 

426 await self._connect_once() 

427 # Normal SSE return (server closed the stream) → retry. 

428 backoff = 1.0 

429 except asyncio.CancelledError: 

430 raise 

431 except Exception as e: 

432 logger.warning( 

433 "connection error: %r; reconnecting in %.1fs", 

434 e, 

435 backoff, 

436 ) 

437 await self._teardown_session() 

438 if self._stopping: 

439 return 

440 try: 

441 await asyncio.sleep(backoff) 

442 except asyncio.CancelledError: 

443 raise 

444 backoff = min(backoff * 2.0, 30.0)

446 async def _connect_once(self) -> None: 

447 url = f"{self._central_url}/events" 

448 headers = { 

449 "Authorization": f"Bearer {self._hf_token}", 

450 "Accept": "text/event-stream", 

451 } 

452 # No total timeout (long-lived); sock_read covers stalls. 

453 timeout = aiohttp.ClientTimeout(total=None, sock_read=60.0) 

454 assert self._http is not None 

455 async with self._http.get(url, headers=headers, timeout=timeout) as resp: 

456 if resp.status != 200: 

457 txt = (await resp.text())[:200] 

458 raise RuntimeError(f"SSE /events HTTP {resp.status}: {txt!r}") 

459 logger.info("SSE connected to %s", url) 

460 self._sse_resp = resp 

461 try: 

462 async for event in self._iter_sse_events(resp): 

463 if self._stopping: 

464 return 

465 await self._handle_event(event) 

466 finally: 

467 self._sse_resp = None 

468 self._self_peer_id = None

485 @staticmethod 

486 async def _iter_sse_events( 

487 resp: aiohttp.ClientResponse, 

488 ) -> AsyncIterator[dict[str, Any]]: 

489 """Yield decoded JSON dicts from an SSE response.""" 

490 data_lines: List[str] = [] 

491 while True: 

492 raw = await resp.content.readline() 

493 if not raw: 

494 return 

495 line = raw.decode("utf-8", errors="replace").rstrip("\r\n") 

496 if line == "": 

497 if data_lines: 

498 payload = "\n".join(data_lines) 

499 data_lines = [] 

500 try: 

501 yield json.loads(payload) 

502 except json.JSONDecodeError: 

503 logger.debug("non-json SSE data: %r", payload[:200]) 

504 continue 

505 if line.startswith(":"): 

506 continue 

507 if line.startswith("data:"): 

508 data_lines.append(line[5:].lstrip())

513 async def _handle_event(self, msg: dict[str, Any]) -> None: 

514 t = msg.get("type") 

515 if t == "welcome": 

516 self._self_peer_id = msg.get("peerId") 

517 lease = msg.get("lease_seconds") 

518 rec = msg.get("recommended_heartbeat_interval_seconds") 

519 if isinstance(rec, (int, float)) and rec > 0: 

520 self._heartbeat_interval = max(2.0, float(rec)) 

521 elif isinstance(lease, (int, float)) and lease > 0: 

522 self._heartbeat_interval = max(2.0, float(lease) / 3.0) 

523 logger.info( 

524 "welcome peerId=%s (heartbeat %.1fs)", 

525 self._self_peer_id, 

526 self._heartbeat_interval, 

527 ) 

528 await self._post( 

529 { 

530 "type": "setPeerStatus", 

531 "roles": ["listener"], 

532 "meta": {"name": self._consumer_label}, 

533 } 

534 ) 

535 # Keep our listener peer registered: central evicts peers with 

536 # no inbound /send within the lease (SSE pings don't refresh 

537 # it), which otherwise leaves a zombie SSE + startSession 400s. 

538 self._ensure_heartbeat_task() 

539 # Session establishment is handled solely by the HTTP-poll loop 

540 # (_discovery_loop), for both pinned and auto-picked targets. 

541 # central's SSE `list` push is unreliable for already-online 

542 # producers, and routing startSession through the loop keeps it 

543 # off the SSE-read path — so a failed startSession (e.g. the 

544 # producer restarted with a new peerId) is retried gracefully 

545 # instead of tearing down the SSE connection. 

546 self._ensure_discovery_task() 

547 elif t == "list": 

548 producers = msg.get("producers") or [] 

549 names = [(p.get("meta") or {}).get("name") for p in producers] 

550 logger.debug( 

551 "list event (informational): %d producers names=%r", 

552 len(producers), 

553 names, 

554 ) 

555 elif t == "peer": 

556 sdp = msg.get("sdp") 

557 ice = msg.get("ice") 

558 if sdp: 

559 await self._handle_remote_sdp(sdp) 

560 if ice: 

561 await self._handle_remote_ice(ice) 

562 elif t == "endSession": 

563 reason = msg.get("reason") 

564 sid = msg.get("sessionId") 

565 # Central also pushes endSession for *stale* sessions — e.g. 

566 # a previous consumer instance (same HF identity) that died 

567 # without cleaning up. Those carry a sessionId that doesn't 

568 # match anything in this process, so we just acknowledge and 

569 # keep the SSE open instead of tearing down and reconnecting. 

570 if self._session_id is not None and sid == self._session_id: 

571 logger.info("endSession for our session %s (reason=%r)", sid, reason) 

572 raise RuntimeError(f"endSession: {reason}") 

573 logger.debug( 

574 "ignoring stale endSession sessionId=%s reason=%r (we hold session=%r)", 

575 sid, 

576 reason, 

577 self._session_id, 

578 ) 

579 elif t == "sessionRejected": 

580 reason = msg.get("reason") 

581 logger.warning("sessionRejected (reason=%s)", reason) 

582 raise RuntimeError(f"sessionRejected: {reason}") 

583 else: 

584 logger.debug("unhandled event type=%r", t)

590 def _auto_pick_producer(self, producers: list[dict[str, Any]]) -> None: 

591 target = self._canonical_name(self._robot_name) 

592 # Canonicalised name match — "reachy_mini", "reachy-mini", 

593 # "ReachyMini" all match "reachymini". 

594 for p in producers: 

595 meta = p.get("meta") or {} 

596 if self._canonical_name(meta.get("name")) == target: 

597 self._target_peer_id = p.get("id") 

598 logger.info( 

599 "selected robot peerId=%s name=%r transport=%r", 

600 self._target_peer_id, 

601 meta.get("name"), 

602 meta.get("transport"), 

603 ) 

604 return 

605 # Fallback: if there's only one producer for this token, use it 

606 # regardless of how it named itself. Pin REACHY_ROBOT_PEER_ID if 

607 # this guess is ever wrong. 

608 if len(producers) == 1: 

609 only = producers[0] 

610 meta = only.get("meta") or {} 

611 self._target_peer_id = only.get("id") 

612 logger.info( 

613 "only one producer visible (name=%r) and name didn't match " 

614 "%r — picking it anyway. Set REACHY_ROBOT_PEER_ID to pin a specific id.", 

615 meta.get("name"), 

616 self._robot_name, 

617 ) 

618 return 

619 names = [(p.get("meta") or {}).get("name") for p in producers] 

620 logger.info( 

621 "no producer matched name=%r (canonical=%r) among %d producers %r; " 

622 "waiting for next list", 

623 self._robot_name, 

624 target, 

625 len(producers), 

626 names, 

627 )

638 async def _heartbeat_loop(self) -> None: 

639 """Refresh our peer lease so central doesn't evict the listener. 

640 

641 Central only refreshes ``last_seen`` on inbound /send (SSE pings 

642 don't count), so an idle listener gets swept after the lease and 

643 its SSE becomes a zombie. A periodic setPeerStatus keeps us 

644 registered. 

645 """ 

646 while not self._stopping and self._sse_resp is not None: 

647 try: 

648 await asyncio.sleep(self._heartbeat_interval) 

649 if self._stopping or self._sse_resp is None: 

650 return 

651 await self._post( 

652 { 

653 "type": "setPeerStatus", 

654 "roles": ["listener"], 

655 "meta": {"name": self._consumer_label}, 

656 } 

657 ) 

658 except asyncio.CancelledError: 

659 raise 

660 except Exception as e: 

661 logger.warning("heartbeat error: %r", e)

665 def _ensure_discovery_task(self) -> None: 

666 """Start the HTTP-poll discovery loop once per SSE connection. 

667 

668 Runs for pinned targets too: the loop skips the robot-status poll 

669 while a target is set and goes straight to startSession, but it can 

670 also re-discover by name if that target turns out to be stale (e.g. 

671 the robot restarted with a new peerId). 

672 """ 

673 if self._discovery_task is not None and not self._discovery_task.done(): 

674 return 

675 self._discovery_task = asyncio.create_task( 

676 self._discovery_loop(), name="reachy-discovery" 

677 )

679 async def _robot_status(self) -> list[dict[str, Any]]: 

680 """GET {central}/api/robot-status — the reliable producer list. 

681 

682 The SSE `list` push only re-fires on producer status *changes*, so 

683 a robot that was already online before we connected can be absent 

684 from the SSE stream entirely. This HTTP endpoint always returns the 

685 current set, so we use it for discovery. 

686 """ 

687 assert self._http is not None 

688 url = f"{self._central_url}/api/robot-status" 

689 headers = {"Authorization": f"Bearer {self._hf_token}"} 

690 async with self._http.get( 

691 url, 

692 headers=headers, 

693 timeout=aiohttp.ClientTimeout(total=10.0), 

694 ) as resp: 

695 resp.raise_for_status() 

696 payload = await resp.json() 

697 return payload.get("robots") or [] 

698 

699 @staticmethod 

700 def _normalize_robot_status( 

701 robots: list[dict[str, Any]], 

702 ) -> list[dict[str, Any]]: 

703 """Normalise /api/robot-status entries to the SSE `list` shape. 

704 

705 Maps each entry to ``{id, meta}`` so `_auto_pick_producer` can 

706 consume either source. 

707 """ 

708 out = [] 

709 for r in robots: 

710 meta = r.get("meta") or {} 

711 if "name" not in meta and r.get("robotName"): 

712 meta = {**meta, "name": r.get("robotName")} 

713 out.append({"id": r.get("peerId") or r.get("id"), "meta": meta}) 

714 return out

716 async def _discovery_loop(self) -> None: 

717 """Poll /api/robot-status until we hold a session. 

718 

719 Loops on `_session_id is None` (not `_target_peer_id`) so that a 

720 failed startSession doesn't strand us: on failure we clear the 

721 (unpinned) target and re-discover on the next pass. 

722 """ 

723 delay = 2.0 

724 while not self._stopping and self._session_id is None: 

725 try: 

726 if self._target_peer_id is None: 

727 robots = await self._robot_status() 

728 names = [ 

729 (r.get("meta") or {}).get("name") or r.get("robotName") 

730 for r in robots 

731 ] 

732 logger.debug( 

733 "robot-status poll: %d robots names=%r", 

734 len(robots), 

735 names, 

736 ) 

737 if robots: 

738 self._auto_pick_producer(self._normalize_robot_status(robots)) 

739 if self._target_peer_id is not None and self._session_id is None: 

740 await self._start_session() 

741 except asyncio.CancelledError: 

742 raise 

743 except Exception as e: 

744 logger.warning("discovery error: %r", e) 

745 # Re-discover next pass unless the robot id was pinned. 

746 if self._target_peer_id_pinned is None: 

747 self._target_peer_id = None 

748 if self._session_id is not None: 

749 return 

750 await asyncio.sleep(delay) 

751 delay = min(delay * 1.5, 10.0)

755 async def _post(self, body: dict[str, Any]) -> Optional[dict[str, Any]]: 

756 url = f"{self._central_url}/send" 

757 headers = { 

758 "Authorization": f"Bearer {self._hf_token}", 

759 "Content-Type": "application/json", 

760 } 

761 timeout = aiohttp.ClientTimeout(total=15.0) 

762 assert self._http is not None 

763 try: 

764 async with self._http.post( 

765 url, 

766 headers=headers, 

767 json=body, 

768 timeout=timeout, 

769 ) as resp: 

770 if resp.status != 200: 

771 txt = (await resp.text())[:200] 

772 logger.warning( 

773 "POST /send type=%s -> HTTP %d: %s", 

774 body.get("type"), 

775 resp.status, 

776 txt, 

777 ) 

778 # Central deregistered our peer but our SSE is a zombie. 

779 # Force a fresh SSE connection (new welcome re-registers). 

780 if resp.status == 400 and "Connect to /events" in txt: 

781 self._force_sse_reconnect("peer deregistered") 

782 return None 

783 try: 

784 return cast("Optional[dict[str, Any]]", await resp.json()) 

785 except Exception: 

786 return None 

787 except Exception as e: 

788 logger.warning("POST /send type=%s failed: %r", body.get("type"), e) 

789 return None

791 async def _start_session(self) -> None: 

792 if self._target_peer_id is None: 

793 return 

794 # Guard set synchronously (before any await) so concurrent callers 

795 # can't both pass the check and double-start — that races on 

796 # central and gets one session killed. 

797 if self._session_id is not None or self._starting_session: 

798 return 

799 self._starting_session = True 

800 try: 

801 resp = await self._post( 

802 { 

803 "type": "startSession", 

804 "peerId": self._target_peer_id, 

805 } 

806 ) 

807 if resp is None: 

808 # Transient (central unreachable / deregistered our peer). 

809 # Keep the target and let the discovery loop retry it. 

810 raise RuntimeError("startSession: no response from central") 

811 if resp.get("type") == "sessionStarted": 

812 self._session_id = resp.get("sessionId") 

813 logger.info("sessionStarted sessionId=%s", self._session_id) 

814 # The robot will now push an SDP offer to us via SSE. 

815 return 

816 # Any other response means the producer we targeted is unusable: 

817 # gone (robot restarted with a new peerId), busy, or rejected. 

818 # Drop the stale target so the discovery loop re-picks the robot 

819 # by name on its next poll instead of hammering a dead peerId 

820 # forever. Not raised — a stale pin is a recoverable condition. 

821 detail = resp.get("reason") or resp.get("details") or resp.get("type") 

822 logger.warning( 

823 "startSession failed (peerId=%s reason=%r); re-discovering by name", 

824 self._target_peer_id, 

825 detail, 

826 ) 

827 self._target_peer_id = None 

828 finally: 

829 self._starting_session = False

833 def _spawn_track_task(self, coro: Coroutine[Any, Any, None]) -> None: 

834 """Run a track consumer, holding a reference for its whole life. 

835 

836 The event loop only keeps a weak reference to a running task, so a 

837 bare ``create_task`` can be collected mid-flight and the track 

838 silently stops being consumed. Discarding on completion keeps the 

839 set from growing across reconnects. 

840 """ 

841 task = asyncio.create_task(coro) 

842 self._track_tasks.add(task) 

843 task.add_done_callback(self._track_tasks.discard)

938 async def _handle_remote_sdp(self, sdp_msg: dict[str, Any]) -> None: 

939 sdp_type = sdp_msg.get("type") 

940 sdp_text = sdp_msg.get("sdp") 

941 if not sdp_text or sdp_type != "offer": 

942 logger.warning("unexpected sdp envelope: type=%r", sdp_type) 

943 return 

944 if self._pc is None: 

945 await self._build_pc() 

946 assert self._pc is not None 

947 offer = RTCSessionDescription(sdp=sdp_text, type=sdp_type) 

948 await self._pc.setRemoteDescription(offer) 

949 self._remote_desc_set = True 

950 # Attach our outbound audio track to the audio transceiver the robot's 

951 # (sendrecv) offer just created — after setRemoteDescription and before 

952 # createAnswer so it lands on the existing m-line instead of spawning a 

953 # second one. 

954 if self._out_track is not None and not self._out_track_added: 

955 try: 

956 self._pc.addTrack(self._out_track) 

957 self._out_track_added = True 

958 logger.info("added outbound audio track") 

959 except Exception as e: 

960 logger.warning("addTrack(out) failed: %r", e) 

961 # Drain any ICE candidates that arrived before the offer. 

962 pending, self._pending_remote_ice = self._pending_remote_ice, [] 

963 for c in pending: 

964 await self._add_remote_ice(c) 

965 answer = await self._pc.createAnswer() 

966 await self._pc.setLocalDescription(answer) 

967 # aiortc has finished local ICE gathering at this point; the answer 

968 # SDP carries all of our candidates. We don't trickle outbound ICE. 

969 local_sdp = self._pc.localDescription.sdp 

970 self._log_candidate_types("local", local_sdp) 

971 self._log_candidate_types("remote", sdp_text) 

972 await self._post( 

973 { 

974 "type": "peer", 

975 "sessionId": self._session_id, 

976 "sdp": { 

977 "type": self._pc.localDescription.type, 

978 "sdp": local_sdp, 

979 }, 

980 } 

981 ) 

982 logger.info("sent SDP answer for session %s", self._session_id)

984 @staticmethod 

985 def _log_candidate_types(label: str, sdp: str) -> None: 

986 import re 

987 from collections import Counter 

988 

989 # Candidate line: "candidate:<foundation> <comp> <proto> <prio> <ip> <port> typ <type> ..." 

990 types = Counter(re.findall(r"candidate:[^\r\n]* typ (\w+)", sdp or "")) 

991 n = sum(types.values()) 

992 logger.debug("%s ICE candidates: %d total by type %s", label, n, dict(types))

994 async def _handle_remote_ice(self, ice: dict[str, Any]) -> None: 

995 if not self._remote_desc_set or self._pc is None: 

996 self._pending_remote_ice.append(ice) 

997 return 

998 await self._add_remote_ice(ice)

1000 async def _add_remote_ice(self, ice: dict[str, Any]) -> None: 

1001 cand_str = (ice.get("candidate") or "").strip() 

1002 if not cand_str: 

1003 return # end-of-candidates marker; aiortc doesn't need it 

1004 try: 

1005 sdp_cand = cand_str 

1006 prefix = "candidate:" 

1007 if sdp_cand.startswith(prefix): 

1008 sdp_cand = sdp_cand[len(prefix) :] 

1009 candidate = candidate_from_sdp(sdp_cand) 

1010 candidate.sdpMid = ice.get("sdpMid") 

1011 mline = ice.get("sdpMLineIndex") 

1012 if mline is not None: 

1013 candidate.sdpMLineIndex = int(mline) 

1014 assert self._pc is not None 

1015 await self._pc.addIceCandidate(candidate) 

1016 # Log the type so we can see what the robot offers us. 

1017 import re 

1018 

1019 m = re.search(r" typ (\w+)", cand_str) 

1020 logger.debug( 

1021 "+remote ICE %s: %s", 

1022 m.group(1) if m else "?", 

1023 cand_str[:80], 

1024 ) 

1025 except Exception as e: 

1026 logger.warning("addIceCandidate failed: %r (cand=%r)", e, cand_str)

1028 async def _consume_video(self, track: MediaStreamTrack) -> None: 

1029 try: 

1030 while True: 

1031 try: 

1032 # recv() is typed as the Frame|Packet union; _on_track 

1033 # only starts this for kind == "video", so aiortc yields 

1034 # VideoFrames here. 

1035 frame = cast("av.VideoFrame", await track.recv()) 

1036 except MediaStreamError: 

1037 # Normal end-of-track: aiortc raises this on teardown. 

1038 logger.debug("video track ended") 

1039 return 

1040 try: 

1041 # to_ndarray's dtype depends on the format; rgb24 is 

1042 # 8 bits per channel by definition, which is what 

1043 # latest_frame() promises its callers. 

1044 arr = cast( 

1045 "npt.NDArray[np.uint8]", frame.to_ndarray(format="rgb24") 

1046 ) 

1047 except Exception as e: 

1048 logger.warning("frame decode failed: %r", e) 

1049 continue 

1050 # Atomic reference swap of the (id, ndarray) tuple — readers 

1051 # see either the prior pair or this one, never a mismatch. 

1052 self._frame_counter += 1 

1053 self._latest = (self._frame_counter, arr) 

1054 # Source resolution as the daemon's webrtcsink emits it 

1055 # (h, w) — consumers read this to pick the right camera 

1056 # crop_scale when computing look-at geometry. 

1057 if self._source_size is None or self._source_size != ( 

1058 arr.shape[1], 

1059 arr.shape[0], 

1060 ): 

1061 self._source_size = (arr.shape[1], arr.shape[0]) 

1062 logger.info( 

1063 "source frame size: %dx%d", 

1064 self._source_size[0], 

1065 self._source_size[1], 

1066 ) 

1067 except asyncio.CancelledError: 

1068 raise 

1069 except Exception as e: 

1070 # Terminal for this track: nothing restarts the consumer, so the 

1071 # traceback is the only clue left. `exception` logs it at error. 

1072 logger.exception("video consumer stopped: %r", e)

1074 async def _consume_audio(self, track: MediaStreamTrack) -> None: 

1075 """Decode the robot mic to mono float32 and hand chunks to ``on_pcm``. 

1076 

1077 aiortc delivers Opus as 48 kHz s16 ``av.AudioFrame``s; the resampler 

1078 downmixes to mono and converts to ``flt`` at ``audio_sample_rate``. 

1079 """ 

1080 resampler = av.audio.resampler.AudioResampler( 

1081 format="flt", 

1082 layout="mono", 

1083 rate=self._audio_sample_rate, 

1084 ) 

1085 try: 

1086 while True: 

1087 try: 

1088 # Same union as in _consume_video; _on_track only starts 

1089 # this for kind == "audio". 

1090 frame = cast("av.AudioFrame", await track.recv()) 

1091 except MediaStreamError: 

1092 # Normal end-of-track: aiortc raises this on teardown. 

1093 logger.debug("audio track ended") 

1094 return 

1095 try: 

1096 for out in resampler.resample(frame): 

1097 # "flt"/mono → ndarray shape (1, samples); flatten. 

1098 arr = out.to_ndarray().reshape(-1) 

1099 pcm = arr.astype(np.float32, copy=False) 

1100 if pcm.size == 0: 

1101 continue 

1102 self._audio_frames += 1 

1103 self._pcm_samples += int(pcm.size) 

1104 if self._on_pcm is not None: 

1105 self._on_pcm(pcm) 

1106 except Exception as e: 

1107 logger.warning("audio frame decode failed: %r", e) 

1108 continue 

1109 except asyncio.CancelledError: 

1110 raise 

1111 except Exception as e: 

1112 # Terminal for this track (see _consume_video). 

1113 logger.exception("audio consumer stopped: %r", e)

1117 async def _teardown_session(self) -> None: 

1118 # Best-effort endSession to central before tearing down the pc. 

1119 # Only if we actually held a session AND the SSE is still alive 

1120 # — otherwise central rejects with HTTP 400 "Connect to /events 

1121 # first" and that's just noise in the logs. 

1122 if ( 

1123 self._session_id is not None 

1124 and self._http is not None 

1125 and not self._http.closed 

1126 ): 

1127 try: 

1128 resp = await self._post( 

1129 { 

1130 "type": "endSession", 

1131 "sessionId": self._session_id, 

1132 } 

1133 ) 

1134 if resp is None: 

1135 # POST already logged its own warning, no need to echo. 

1136 pass 

1137 except Exception: 

1138 pass 

1139 self._session_id = None 

1140 self._starting_session = False 

1141 self._remote_desc_set = False 

1142 self._pending_remote_ice = [] 

1143 self._pc_connected = False 

1144 # Drop the last decoded frame so latest_frame() doesn't keep 

1145 # handing out a stale image that looks live after a disconnect. 

1146 # frame_id keeps climbing across reconnects, so pollers still see 

1147 # fresh ids once a new session delivers frames. 

1148 self._latest = None 

1149 self._source_size = None 

1150 if self._pc is not None: 

1151 try: 

1152 await self._pc.close() 

1153 except Exception: 

1154 pass 

1155 self._pc = None 

1156 # Closing the pc ends the tracks, so the consumers normally exit on 

1157 # their own MediaStreamError. Cancel anyway: a consumer blocked 

1158 # elsewhere would otherwise outlive the session it belongs to and 

1159 # keep feeding on_pcm from a connection the caller thinks is gone. 

1160 for task in list(self._track_tasks): 

1161 task.cancel() 

1162 self._track_tasks.clear() 

1163 # If the robot peerId wasn't pinned via env, force a fresh pick on 

1164 # the next list event in case the robot reconnected with a new id. 

1165 if self._target_peer_id_pinned is None: 

1166 self._target_peer_id = None
