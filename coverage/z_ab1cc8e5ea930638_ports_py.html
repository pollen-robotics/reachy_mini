1"""Serial port discovery for the NFC reader board. 

2 

3The board is recognised by its USB product string, ``Winnie NFC <version>``. 

4Reading it is left to the driver, ``winnie_nfc.usb``: pyserial reports it on 

5Linux and macOS, and the driver reads it from the OS on Windows. 

6""" 

7 

8from __future__ import annotations 

9 

10import logging 

11from dataclasses import dataclass 

12from types import ModuleType 

13 

14logger = logging.getLogger(__name__) 

15 

16# USB product string of the NFC board, followed by its version: "Winnie NFC 1.0". 

17NFC_PRODUCT_PREFIX = "Winnie NFC" 

18 

19 

20@dataclass(frozen=True) 

21class NfcBoard: 

22 """An NFC reader board found on a serial port.""" 

23 

24 port: str 

25 version: str | None # from the USB product string; None if it has none 

26 

27 

28def parse_product(product: str | None) -> tuple[bool, str | None]: 

29 """Tell whether a USB product string is the NFC board's, and its version. 

30 

31 Returns ``(is_nfc_board, version)``. The version is whatever follows the 

32 prefix, surrounding blanks removed: ``"Winnie NFC 1.2"`` gives ``"1.2"``; 

33 a bare ``"Winnie NFC"`` is the board with no version. 

34 """ 

35 if not product: 

36 return False, None 

37 product = product.strip() 

38 if not product.casefold().startswith(NFC_PRODUCT_PREFIX.casefold()): 

39 return False, None 

40 version = product[len(NFC_PRODUCT_PREFIX) :].strip() 

41 return True, version or None 

42 

43 

44def _usb() -> ModuleType | None: 

45 """Return the driver's USB module, or None when it cannot be imported. 

46 

47 Port lookup must never take the daemon down: the motor port search goes 

48 through here too (see ``daemon.utils.find_serial_port``). 

49 """ 

50 try: 

51 from winnie_nfc import usb 

52 except Exception as e: # noqa: BLE001 - a broken install reads as no board 

53 logger.debug("winnie_nfc unavailable, no NFC board lookup: %s", e) 

54 return None 

55 module: ModuleType = usb # untyped package: name what it is 

56 return module 

57 

58 

59def nfc_board_ports() -> list[str]: 

60 """Return the ports of every NFC reader board plugged in, sorted.""" 

61 usb = _usb() 

62 if usb is None: 

63 return [] 

64 found: list[str] = usb.winnie_ports() 

65 return found 

66 

67 

68def find_nfc_board() -> NfcBoard | None: 

69 """Return the NFC reader board, or ``None`` if there is none. 

70 

71 With several boards plugged in, the first port in sort order. 

72 """ 

73 found = nfc_board_ports() 

74 if len(found) > 1: 

75 logger.warning( 

76 "Several NFC reader boards found (%s), using %s.", found, found[0] 

77 ) 

78 return describe_port(found[0]) if found else None 

79 

80 

81def describe_port(port: str) -> NfcBoard: 

82 """Describe a port, with the board version if its product string has one.""" 

83 usb = _usb() 

84 info = usb.port_info(port) if usb is not None else None 

85 product: str | None = getattr(info, "product", None) 

86 return NfcBoard(port=port, version=parse_product(product)[1])
