17# --- TURN relay support ---------------------------------------------- 

18# 

19# By default webrtcsink only gathers host + srflx candidates, so a remote 

20# consumer behind a restrictive NAT (e.g. a cloud backend that can't 

21# UDP-hole-punch) has no way to reach the robot. Short-lived Cloudflare 

22# TURN credentials from HF's hosted proxy — authenticated with the 

23# daemon's own HF token, the same one the central relay uses — let the 

24# robot offer a ``relay`` candidate as well. 

25# 

26# Only the robot has to *offer* a relay: a consumer reaches it with plain 

27# STUN. So there is no central-server change and no consumer-side 

28# credential to manage (which matters because aiortc's STUN client works 

29# while its TURN client does not). 

30TURN_CREDENTIALS_URL = os.getenv( 

31 "REACHY_TURN_URL", "https://turn.fastrtc.org/credentials" 

32) 

33TURN_TTL_SECONDS = int(os.getenv("REACHY_TURN_TTL", "600")) 

34 

35# Credentials are short-lived; re-fetch well before they expire. 

36_TURN_REFRESH_RATIO = 0.5 

37# A daemon that boots without a network shouldn't wait out a full period. 

38_TURN_RETRY_AFTER_FAILURE_S = 30.0 

39_TURN_HTTP_TIMEOUT_S = 10.0

42def ice_servers_to_turn_uris(servers: List[Dict[str, Any]]) -> List[str]: 

43 """Convert ICE-server dicts to webrtcbin ``turn(s)://user:pass@host:port`` URIs. 

44 

45 ``stun:`` entries and entries without credentials are skipped — 

46 webrtcbin takes a STUN server through its own ``stun-server`` property, 

47 and ``add-turn-server`` rejects a URI with no userinfo. 

48 

49 Args: 

50 servers: ``{"urls", "username"?, "credential"?}`` dicts as returned 

51 by the TURN credentials proxy. ``urls`` may be a single string 

52 or a list of strings. 

53 

54 Returns: 

55 One URI per TURN url, with credentials percent-encoded so a 

56 password containing ``:``, ``@`` or ``/`` can't corrupt the URI. 

57 

58 """ 

59 uris: List[str] = [] 

60 for server in servers: 

61 urls = server.get("urls") 

62 if not urls: 

63 continue 

64 if isinstance(urls, str): 

65 urls = [urls] 

66 user = server.get("username") 

67 cred = server.get("credential") 

68 if user is None or cred is None: 

69 continue 

70 auth = f"{quote(str(user), safe='')}:{quote(str(cred), safe='')}" 

71 for url in urls: 

72 scheme, _, rest = url.partition(":") 

73 scheme = scheme.lower() 

74 if scheme in ("turn", "turns") and rest: 

75 uris.append(f"{scheme}://{auth}@{rest}") 

76 return uris

79def _turn_hosts(uris: List[str]) -> List[str]: 

80 """Strip userinfo from TURN URIs so they can be logged safely.""" 

81 return [u.split("@", 1)[-1] for u in uris]

84class TurnCredentials: 

85 """Cloudflare TURN credentials kept fresh by a background thread. 

86 

87 :meth:`turn_uris` never blocks and never raises: it returns whatever 

88 the refresher last fetched successfully. That is the whole point of 

89 the class — its only caller runs inside GStreamer's ``consumer-added`` 

90 signal, where the SDP offer for that consumer cannot be generated 

91 until the handler returns. Fetching there would delay *every* client, 

92 including LAN ones that will never use a relay, by however long the 

93 TURN proxy takes to answer. 

94 

95 Credentials expire, so the thread re-fetches at half their TTL, and 

96 sooner than that after a failure. 

97 """ 

98 

99 def __init__( 

100 self, 

101 url: str = TURN_CREDENTIALS_URL, 

102 ttl: int = TURN_TTL_SECONDS, 

103 ) -> None: 

104 """Build a refresher. Nothing is fetched until :meth:`start`. 

105 

106 Args: 

107 url: TURN credentials proxy endpoint. 

108 ttl: lifetime in seconds to request for the credentials; the 

109 refresh period is half this. 

110 

111 """ 

112 self._url = url 

113 self._ttl = ttl 

114 # Written only by the refresher thread, read by the GStreamer 

115 # thread. Rebound as a whole list, never mutated in place, so a 

116 # reader sees either the previous set or the new one. 

117 self._uris: List[str] = [] 

118 self._warned_no_token = False 

119 self._stop = threading.Event() 

120 self._thread: Optional[threading.Thread] = None 

121 

122 def start(self) -> None: 

123 """Start the refresher thread. Idempotent.""" 

124 if self._thread is not None and self._thread.is_alive(): 

125 return 

126 self._stop.clear() 

127 self._thread = threading.Thread( 

128 target=self._refresh_loop, name="turn-credentials", daemon=True 

129 ) 

130 self._thread.start() 

131 

132 def stop(self) -> None: 

133 """Ask the refresher thread to exit. Does not join.""" 

134 self._stop.set() 

135 

136 def turn_uris(self) -> List[str]: 

137 """Return the TURN URIs last fetched, or ``[]`` if none ever were.""" 

138 return self._uris 

139 

140 def _refresh_loop(self) -> None: 

141 while not self._stop.is_set(): 

142 self._stop.wait(self._refresh_once()) 

143 

144 def _refresh_once(self) -> float: 

145 """Refresh the cache; return how long to wait before trying again. 

146 

147 Only a transient failure earns the short retry. A robot that is 

148 simply not logged in has nothing to retry *for*, and this runs for 

149 the daemon's whole life — backing off fast there would put a 

150 warning in the log every 30 seconds forever. 

151 """ 

152 period = self._ttl * _TURN_REFRESH_RATIO 

153 try: 

154 from huggingface_hub import get_token 

155 

156 token = get_token() 

157 if not token: 

158 # Steady state on a robot nobody has logged in; say so once. 

159 if not self._warned_no_token: 

160 self._warned_no_token = True 

161 logger.info( 

162 "No HF token; not offering TURN relay candidates. " 

163 "Log in to reach this robot from outside its network." 

164 ) 

165 return period 

166 resp = requests.get( 

167 self._url, 

168 headers={"Authorization": f"Bearer {token}"}, 

169 params={"ttl": self._ttl}, 

170 timeout=_TURN_HTTP_TIMEOUT_S, 

171 ) 

172 resp.raise_for_status() 

173 uris = ice_servers_to_turn_uris(resp.json().get("iceServers") or []) 

174 except Exception as e: # noqa: BLE001 - a relay is best-effort 

175 logger.warning("Failed to fetch TURN credentials: %r", e) 

176 return _TURN_RETRY_AFTER_FAILURE_S 

177 self._warned_no_token = False 

178 if not uris: 

179 # The proxy answered but offered no relay: not worth hammering. 

180 logger.info("TURN proxy returned no relay servers") 

181 return period 

182 self._uris = uris 

183 logger.info("Refreshed %d TURN server(s): %s", len(uris), _turn_hosts(uris)) 

184 return period

187def get_producer_list(host: str, port: int) -> Dict[str, Dict[str, str]]: 

188 """Get the list of gstreamer producers from the signalling server. 

189 

190 Args: 

191 host (str): The hostname or IP address of the signalling server. 

192 port (int): The port number of the signalling server. 

193 

194 Returns: 

195 Dict[str, Dict[str, str]]: A dictionary mapping producer IDs to their metadata dictionaries. 

196 

197 """ 

198 with connect(f"ws://{host}:{port}") as websocket: 

199 _ = websocket.recv() # welcome message is ignored 

200 message = json.dumps({"type": "list"}) 

201 websocket.send(message) 

202 message = json.loads(websocket.recv()) 

203 logging.debug(f"Received: {message}") 

204 if message.get("type") == "list": 

205 producers = {p["id"]: p["meta"] for p in message.get("producers", [])} 

206 return producers 

207 else: 

208 logging.warning(f"Received unknown message type: {message}.") 

209 return {}

212def find_producer_peer_id_by_name(host: str, port: int, name: str) -> str: 

213 """Find the peer ID of a producer by its name. 

214 

215 Args: 

216 host: Host address of the signalling server. 

217 port: Port number of the signalling server. 

218 name: Producer name to search for. 

219 

220 Returns: 

221 Peer ID of the first matching producer. 

222 

223 Raises: 

224 KeyError: If no producer with the specified name is found. 

225 

226 """ 

227 producers = get_producer_list(host=host, port=port) 

228 

229 for producer_id, producer_meta in producers.items(): 

230 if producer_meta["name"] == name: 

231 return producer_id 

232 

233 raise KeyError(f"Producer {name} not found.")

236def main() -> None: 

237 """Get and print the gstreamer producer list.""" 

238 parser = argparse.ArgumentParser(description="Get gstreamer producer list") 

239 parser.add_argument("--signalling-host", default="127.0.0.1") 

240 parser.add_argument("--signalling-port", default=8443, type=int) 

241 parser.add_argument("--verbose", "-v", action="count") 

242 args = parser.parse_args() 

243 

244 if args.verbose: 

245 logging.basicConfig(level=logging.DEBUG) 

246 

247 producers = get_producer_list(args.signalling_host, args.signalling_port) 

248 

249 if producers: 

250 print("List received, producers:") 

251 for producer_id, producer_meta in producers.items(): 

252 print(f" - {producer_id}: {producer_meta}") 

253 else: 

254 print("List received, no producers.")

258 main()
