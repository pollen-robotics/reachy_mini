1"""Persisted daemon config for the startup app. 

2 

3The startup app (launched on the robot's first wake-up) is stored in a small 

4JSON file in the user's config dir, so the choice survives reboots and app 

5updates, stays per-user (not shared across OS accounts on one machine), and can 

6be set over the REST API instead of only via a CLI flag. 

7""" 

8 

9import json 

10import logging 

11from pathlib import Path 

12 

13import platformdirs 

14 

15logger = logging.getLogger(__name__) 

16 

17_KEY = "startup_app" 

18_EQ_KEY = "speaker_eq_gains" 

19_TURN_KEY = "turn_enabled" 

20_FIRST_WAKE_UP_KEY = "first_wake_up_completed" 

21# equalizer-10bands accepts per-band gains in [-24, +12] dB. 

22_EQ_GAIN_MIN, _EQ_GAIN_MAX = -24.0, 12.0 

23 

24 

25def _is_valid_gain(value: object) -> bool: 

26 """Return True for a real number within the equalizer dB range. 

27 

28 The range comparison also rejects NaN and infinities (they compare False) 

29 and oversized ints (exact int/float compare, so no OverflowError) without 

30 converting the value. 

31 """ 

32 return ( 

33 isinstance(value, (int, float)) 

34 and not isinstance(value, bool) 

35 and _EQ_GAIN_MIN <= value <= _EQ_GAIN_MAX 

36 ) 

37 

38 

39def _config_path() -> Path: 

40 """Path to the daemon config file in the user's config dir.""" 

41 return Path(platformdirs.user_config_dir("reachy_mini")) / "daemon_config.json" 

42 

43 

44def _read() -> dict: # type: ignore[type-arg] 

45 """Load the config dict, or {} if missing/unreadable (best-effort).""" 

46 path = _config_path() 

47 try: 

48 with path.open() as f: 

49 data = json.load(f) 

50 return data if isinstance(data, dict) else {} 

51 except FileNotFoundError: 

52 return {} 

53 except (OSError, json.JSONDecodeError) as e: 

54 logger.warning(f"Ignoring unreadable daemon config {path}: {e}") 

55 return {} 

56 

57 

58def get_startup_app() -> str | None: 

59 """Return the persisted startup app name, or None if unset.""" 

60 value = _read().get(_KEY) 

61 return value if isinstance(value, str) else None 

62 

63 

64def get_speaker_eq_gains() -> list[float] | None: 

65 """Return the 10 speaker-EQ band gains (dB), or None if unset/invalid. 

66 

67 Invalid values (wrong length, non-numeric, NaN/inf, or outside the 

68 equalizer-10bands [-24, +12] dB range) are treated as unset so the caller 

69 falls back to its built-in default. 

70 """ 

71 config = _read() 

72 if _EQ_KEY not in config: 

73 return None 

74 value = config[_EQ_KEY] 

75 if ( 

76 isinstance(value, list) 

77 and len(value) == 10 

78 and all(_is_valid_gain(x) for x in value) 

79 ): 

80 return [float(x) for x in value] 

81 # Present but malformed: warn so the user knows their values were ignored. 

82 logger.warning( 

83 "Ignoring invalid '%s' in daemon config (need 10 finite dB gains in " 

84 "[%g, %g]); using the built-in defaults.", 

85 _EQ_KEY, 

86 _EQ_GAIN_MIN, 

87 _EQ_GAIN_MAX, 

88 ) 

89 return None 

90 

91 

92def _get_bool(key: str) -> bool | None: 

93 """Return the boolean at `key`, or None when unset or malformed. 

94 

95 A present-but-malformed value is warned about (so the user knows their 

96 hand-edited config was ignored) and treated as unset. 

97 """ 

98 config = _read() 

99 if key not in config: 

100 return None 

101 value = config[key] 

102 if isinstance(value, bool): 

103 return value 

104 logger.warning( 

105 "Ignoring invalid '%s' in daemon config (need true or false); " 

106 "using the built-in default.", 

107 key, 

108 ) 

109 return None 

110

112def get_turn_enabled() -> bool | None: 

113 """Return whether the media server should offer TURN relay candidates. 

114 

115 Returns None if unset or malformed, so the caller keeps its default. A 

116 relay lets a remote consumer behind a restrictive NAT reach the robot; 

117 turning it off saves a background thread refreshing credentials on a 

118 robot that is only ever reached from its own network. 

119 """ 

120 return _get_bool(_TURN_KEY) 

121 

122 

123def _update(key: str, value: object | None) -> None: 

124 """Read-modify-write a single config key; `None` clears it. 

125 

126 Other keys are preserved. May raise OSError on a write error; callers 

127 that must never raise (e.g. the daemon command loop) wrap it themselves. 

128 """ 

129 config = _read() 

130 if value is None: 

131 config.pop(key, None) 

132 else: 

133 config[key] = value 

134 

135 path = _config_path() 

136 path.parent.mkdir(parents=True, exist_ok=True) 

137 # Write to a sibling then rename: a crash mid-write would otherwise 

138 # truncate the file, and the next _update() rebuilds it from the empty 

139 # dict _read() returns for unparseable JSON, dropping every other key. 

140 tmp = path.with_suffix(".json.tmp") 

141 with tmp.open("w") as f: 

142 json.dump(config, f, indent=2) 

143 tmp.replace(path) 

144 

145 

146def set_startup_app(name: str | None) -> None: 

147 """Persist the startup app name; a falsy name clears it.""" 

148 _update(_KEY, name if name else None) 

149 

150 

151def get_first_wake_up_completed() -> bool: 

152 """Return True if the first wake-up setup wizard has been completed. 

153 

154 Robot-wide, persistent flag (not per-session): the mobile / desktop apps 

155 run a one-time, post-connection hardware diagnostic wizard, and gate it on 

156 this so it only ever shows once, whichever client connects. Defaults to 

157 False (wizard pending) when unset or malformed, so a config problem can 

158 never trap the daemon command loop. 

159 """ 

160 return _get_bool(_FIRST_WAKE_UP_KEY) is True 

161 

162 

163def set_first_wake_up_completed(is_completed: bool) -> bool: 

164 """Persist the first wake-up completion flag. Returns True on success. 

165 

166 Fail-safe: a write error is logged and reported as False instead of 

167 raising, so a storage problem can't break the daemon command loop. 

168 """ 

169 try: 

170 _update(_FIRST_WAKE_UP_KEY, bool(is_completed)) 

171 return True 

172 except OSError as e: 

173 logger.warning(f"Could not persist first-wake-up flag: {e}") 

174 return False
