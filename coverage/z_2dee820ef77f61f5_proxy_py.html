1"""Explicit HTTP(S) proxy resolution for aiohttp call sites. 

2 

3Why not ``ClientSession(trust_env=True)``: that flag makes aiohttp honour 

4``HTTP_PROXY``/``HTTPS_PROXY``/``NO_PROXY`` — but it ALSO makes it read 

5``~/.netrc`` for HTTP auth, and aiohttp does not let netrc defer to an 

6explicit ``Authorization`` header: with a netrc entry matching the request 

7host (or a ``default`` entry, which pip/git tooling commonly writes on dev 

8machines), every request that carries a Bearer header raises 

9``ValueError("Cannot combine AUTHORIZATION header with AUTH argument")``, 

10and unauthenticated requests silently gain netrc Basic credentials. Every 

11authenticated Hugging Face / central-relay call in this package sends an 

12explicit Bearer header, so ``trust_env=True`` turns a stray developer 

13``.netrc`` into a hard failure of login, app update checks and remote 

14access. 

15 

16Instead, this module resolves the proxy with the same stdlib machinery 

17``requests`` uses (``urllib.request.getproxies`` / ``proxy_bypass``) and 

18call sites pass it explicitly as aiohttp's per-request ``proxy=`` argument. 

19Environment semantics are identical to ``trust_env`` (scheme-keyed 

20``HTTP_PROXY``/``HTTPS_PROXY``, ``NO_PROXY`` bypass, https-scheme proxy 

21URLs skipped) with zero netrc involvement. 

22 

23Guarded by ``tests/unit_tests/test_proxy_env_support.py``: an AST check 

24fails if a ``ClientSession`` is created with ``trust_env`` or if a session 

25request is made without ``proxy=``. 

26""" 

27 

28from __future__ import annotations 

29 

30import logging 

31from urllib.parse import urlsplit 

32from urllib.request import getproxies, proxy_bypass 

33 

34logger = logging.getLogger(__name__) 

35 

36# aiohttp's per-request `proxy=` only supports plain-HTTP proxy URLs (it 

37# tunnels HTTPS through them with CONNECT). `trust_env=True` silently skips 

38# https-scheme proxy URLs with a warning; we mirror that behaviour instead 

39# of letting aiohttp raise at request time. Warn once, not per request. 

40_warned_unsupported_proxy: set[str] = set() 

41 

42 

43def proxy_for(url: str) -> str | None: 

44 """Return the proxy URL to use for ``url``, or ``None`` to go direct. 

45 

46 Reads ``HTTP_PROXY`` / ``HTTPS_PROXY`` (upper or lower case) via 

47 ``urllib.request.getproxies()`` and honours ``NO_PROXY`` via 

48 ``proxy_bypass()`` — the same stdlib machinery ``requests`` builds on. 

49 The lookup is strictly scheme-keyed, matching aiohttp's own 

50 ``trust_env`` behaviour: ``ALL_PROXY`` is NOT honoured (``requests`` 

51 would fall back to it, aiohttp never did — configure the scheme 

52 variables explicitly). Resolved on every call so an environment change 

53 takes effect without recreating sessions, matching ``trust_env`` 

54 semantics. 

55 

56 Args: 

57 url: The absolute request URL. 

58 

59 Returns: 

60 The proxy URL for the request's scheme, or ``None`` when no proxy 

61 is configured, the host matches ``NO_PROXY``, or the configured 

62 proxy URL uses a scheme aiohttp cannot tunnel through. 

63 

64 """ 

65 try: 

66 parts = urlsplit(url) 

67 host = parts.hostname 

68 except ValueError: 

69 # Malformed URL (e.g. bad IPv6 literal): let the HTTP client produce 

70 # its own error for it; resolving "no proxy" must never raise. 

71 return None 

72 if not host: 

73 return None 

74 try: 

75 # Honours NO_PROXY (and platform bypass rules on macOS/Windows). 

76 if proxy_bypass(host): 

77 return None 

78 except (TypeError, OSError): 

79 # A failed bypass lookup means "don't bypass" — the request still 

80 # goes through the proxy (and over TLS for every call site here). 

81 # Same exception set `requests` tolerates (gaierror is an OSError). 

82 pass 

83 proxy = getproxies().get(parts.scheme) 

84 if not proxy: 

85 return None 

86 proxy_scheme = urlsplit(proxy).scheme 

87 if proxy_scheme != "http": 

88 # Parity with aiohttp's trust_env, which skips non-HTTP proxy URLs 

89 # (per-request `proxy=` would raise ValueError on them instead). 

90 if proxy not in _warned_unsupported_proxy: 

91 _warned_unsupported_proxy.add(proxy) 

92 logger.warning( 

93 "Ignoring %s proxy %r: aiohttp only supports plain-HTTP " 

94 "proxies (HTTPS is tunnelled through them with CONNECT).", 

95 proxy_scheme or "schemeless", 

96 proxy, 

97 ) 

98 return None 

99 return proxy
