1"""Access rules to the NFC reader, shared by the REST and JSON-RPC APIs. 

2 

3Each API translates :class:`NfcDisabledError` into its own error (HTTP 503, 

4``nfc_disabled``). The on/off switch is persisted as ``nfc_enabled`` in 

5``daemon_config.json``. 

6""" 

7 

8from __future__ import annotations 

9 

10import threading 

11 

12from reachy_mini.daemon import startup_app_config 

13from reachy_mini.nfc import ( 

14 DISABLED_ERROR, 

15 NfcReader, 

16 NfcStatus, 

17 NfcTag, 

18 disabled_status, 

19 no_tag, 

20) 

21 

22 

23class NfcDisabledError(Exception): 

24 """The reader is switched off, or the daemon runs without one (--no-nfc).""" 

25 

26 def __init__(self) -> None: 

27 """Carry the same message as the status of a disabled reader.""" 

28 super().__init__(DISABLED_ERROR) 

29 

30 

31def status(reader: NfcReader | None) -> NfcStatus: 

32 """Return the reader status; always answers, even without a reader.""" 

33 return reader.get_status() if reader is not None else disabled_status() 

34 

35 

36def tag(reader: NfcReader | None) -> NfcTag: 

37 """Return the tag on the reader; always answers, even without a reader.""" 

38 return reader.get_tag() if reader is not None else no_tag() 

39 

40 

41def require_reader(reader: NfcReader | None) -> NfcReader: 

42 """Return the reader, or refuse when the daemon runs with ``--no-nfc``.""" 

43 if reader is None: 

44 raise NfcDisabledError() 

45 return reader 

46 

47 

48def require_enabled(reader: NfcReader | None) -> NfcReader: 

49 """Return the reader, or refuse a tag operation while it is off.""" 

50 if reader is None or not reader.is_enabled(): 

51 raise NfcDisabledError() 

52 return reader 

53 

54 

55# Saving and applying the switch happen together. 

56_switch_lock = threading.Lock() 

57 

58 

59def switch_reader(reader: NfcReader, enabled: bool) -> NfcStatus: 

60 """Switch the reader on or off now, and remember it across restarts. 

61 

62 Applies even when the choice cannot be saved (``setting_saved=False``). 

63 Blocking: call it off the event loop. 

64 """ 

65 with _switch_lock: 

66 saved = startup_app_config.set_nfc_enabled(enabled) 

67 if enabled: 

68 reader.start() 

69 else: 

70 reader.stop() 

71 return reader.get_status().model_copy(update={"setting_saved": saved})
